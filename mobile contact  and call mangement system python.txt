contacts = {}
calls = []



def add_contact():
    name = input("Enter name: ")
    phone = input("Enter phone number: ")

    contacts[phone] = name

    print("Contact added successfully!")



def view_contacts():
    print("\n===== CONTACTS =====")

    if len(contacts) == 0:
        print("No contacts found.")
    else:
        for phone, name in contacts.items():
            print("Name:", name)
            print("Phone:", phone)
            print("----------------")



def search_contact():
    phone = input("Enter phone number to search: ")

    if phone in contacts:
        print("Name:", contacts[phone])
        print("Phone:", phone)
    else:
        print("Contact not found!")



def delete_contact():
    phone = input("Enter phone number to delete: ")

    if phone in contacts:
        del contacts[phone]
        print("Contact deleted successfully!")
    else:
        print("Contact not found!")



def make_call():
    phone = input("Enter phone number to call: ")

    if phone in contacts:
        print("Calling", contacts[phone], "...")
        calls.append("Outgoing call to " + contacts[phone] + " (" + phone + ")")
        print("Call recorded.")
    else:
        print("Number not saved in contacts.")
        calls.append("Outgoing call to unknown number (" + phone + ")")



def call_history():
    print("\n===== CALL HISTORY =====")

    if len(calls) == 0:
        print("No calls found.")
    else:
        for call in calls:
            print(call)



while True:

    print("\n================================")
    print(" MOBILE CONTACT & CALL SYSTEM")
    print("================================")
    print("1. Add Contact")
    print("2. View Contacts")
    print("3. Search Contact")
    print("4. Delete Contact")
    print("5. Make Call")
    print("6. Call History")
    print("7. Exit")
    print("================================")

    choice = input("Enter your choice: ")

    if choice == "1":
        add_contact()

    elif choice == "2":
        view_contacts()

    elif choice == "3":
        search_contact()

    elif choice == "4":
        delete_contact()

    elif choice == "5":
        make_call()

    elif choice == "6":
        call_history()

    elif choice == "7":
        print("Thank you!")
        break

    else:
        print("Invalid choice!")
